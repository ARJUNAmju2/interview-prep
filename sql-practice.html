1. Table Creation & Design

★★★ Must Know

Create Tables with Relationships

CREATE TABLE Departments (
    DeptId INT PRIMARY KEY IDENTITY(1,1),
    DeptName VARCHAR(100) NOT NULL
);

CREATE TABLE Employees (
    EmpId INT PRIMARY KEY IDENTITY(1,1),
    Name VARCHAR(100) NOT NULL,
    Email VARCHAR(150) UNIQUE,
    Salary DECIMAL(10,2),
    DeptId INT FOREIGN KEY REFERENCES Departments(DeptId),
    JoinDate DATE DEFAULT GETDATE()
);

Add Constraints Later

ALTER TABLE Employees ADD CONSTRAINT CK_Salary CHECK (Salary > 0);
ALTER TABLE Employees ADD CONSTRAINT DF_JoinDate DEFAULT GETDATE() FOR JoinDate;
💡 Always define PRIMARY KEY, FOREIGN KEY, NOT NULL, UNIQUE, CHECK & DEFAULT constraints — interviewers check this.

2. CRUD Operations

★★★ Must Know

INSERT

INSERT INTO Employees (Name, Email, Salary, DeptId)
VALUES ('Rahul', 'rahul@email.com', 55000, 1);

-- Insert multiple rows
INSERT INTO Employees (Name, Email, Salary, DeptId) VALUES
('Priya', 'priya@email.com', 62000, 2),
('Amit', 'amit@email.com', 48000, 1);

UPDATE with condition

UPDATE Employees SET Salary = Salary * 1.10 WHERE DeptId = 1;
UPDATE Employees SET DeptId = 2 WHERE EmpId = 5;

DELETE vs TRUNCATE

DELETE FROM Employees WHERE EmpId = 3;  -- can rollback, fires triggers
TRUNCATE TABLE Employees;               -- faster, resets identity, no rollback

3. JOINs & Subqueries

★★★ Must Know

INNER JOIN

SELECT e.Name, d.DeptName, e.Salary
FROM Employees e
INNER JOIN Departments d ON e.DeptId = d.DeptId;

LEFT JOIN — Find employees without department

SELECT e.Name
FROM Employees e
LEFT JOIN Departments d ON e.DeptId = d.DeptId
WHERE d.DeptId IS NULL;

Self JOIN — Find employees and their managers

SELECT e.Name AS Employee, m.Name AS Manager
FROM Employees e
LEFT JOIN Employees m ON e.ManagerId = m.EmpId;

Subquery — Employees earning more than average

SELECT Name, Salary FROM Employees
WHERE Salary > (SELECT AVG(Salary) FROM Employees);

Correlated Subquery — Department wise max salary

SELECT Name, Salary, DeptId FROM Employees e
WHERE Salary = (SELECT MAX(Salary) FROM Employees WHERE DeptId = e.DeptId);

4. Aggregate & Ranking Functions

★★★ Must Know

GROUP BY with HAVING

-- Departments with more than 3 employees
SELECT DeptId, COUNT(*) AS EmpCount
FROM Employees
GROUP BY DeptId
HAVING COUNT(*) > 3;

ROW_NUMBER / RANK / DENSE_RANK

SELECT Name, Salary, DeptId,
    ROW_NUMBER() OVER (PARTITION BY DeptId ORDER BY Salary DESC) AS RowNum,
    RANK() OVER (PARTITION BY DeptId ORDER BY Salary DESC) AS Rnk,
    DENSE_RANK() OVER (PARTITION BY DeptId ORDER BY Salary DESC) AS DenseRnk
FROM Employees;

Nth Highest Salary (Most Asked!)

-- 2nd highest salary
SELECT MAX(Salary) FROM Employees
WHERE Salary < (SELECT MAX(Salary) FROM Employees);

-- Nth highest using DENSE_RANK
SELECT Salary FROM (
    SELECT Salary, DENSE_RANK() OVER (ORDER BY Salary DESC) AS Rnk
    FROM Employees
) t WHERE Rnk = 2;
💡 ROW_NUMBER = unique numbers | RANK = skips on tie | DENSE_RANK = no skip on tie

5. Stored Procedures & Functions

★★ Frequent

Stored Procedure

CREATE PROCEDURE GetEmployeesByDept
    @DeptId INT
AS
BEGIN
    SELECT Name, Salary FROM Employees WHERE DeptId = @DeptId;
END;

EXEC GetEmployeesByDept @DeptId = 1;

Stored Procedure with OUTPUT

CREATE PROCEDURE GetEmpCount
    @DeptId INT,
    @Count INT OUTPUT
AS
BEGIN
    SELECT @Count = COUNT(*) FROM Employees WHERE DeptId = @DeptId;
END;

Scalar Function

CREATE FUNCTION GetFullName(@FirstName VARCHAR(50), @LastName VARCHAR(50))
RETURNS VARCHAR(100)
AS
BEGIN
    RETURN @FirstName + ' ' + @LastName;
END;

Table-Valued Function

CREATE FUNCTION GetHighEarners(@MinSalary DECIMAL(10,2))
RETURNS TABLE
AS
RETURN (SELECT Name, Salary FROM Employees WHERE Salary > @MinSalary);

6. Scenario-Based (Interview Favorites)

★★★ Must Know

Find Duplicate Records

SELECT Email, COUNT(*) FROM Employees
GROUP BY Email HAVING COUNT(*) > 1;

Delete Duplicates (Keep one)

WITH CTE AS (
    SELECT *, ROW_NUMBER() OVER (PARTITION BY Email ORDER BY EmpId) AS Rn
    FROM Employees
)
DELETE FROM CTE WHERE Rn > 1;

Department-wise Top Earner

WITH Ranked AS (
    SELECT Name, Salary, DeptId,
        RANK() OVER (PARTITION BY DeptId ORDER BY Salary DESC) AS Rnk
    FROM Employees
)
SELECT * FROM Ranked WHERE Rnk = 1;

Employees who joined in last 30 days

SELECT * FROM Employees WHERE JoinDate >= DATEADD(DAY, -30, GETDATE());

Cumulative/Running Total

SELECT Name, Salary,
    SUM(Salary) OVER (ORDER BY EmpId) AS RunningTotal
FROM Employees;

Pivot — Department wise salary summary

SELECT * FROM (
    SELECT DeptId, Salary FROM Employees
) src
PIVOT (SUM(Salary) FOR DeptId IN ([1],[2],[3])) AS pvt;

7. Indexes & Performance

★★ Frequent

Create Index

CREATE INDEX IX_Emp_DeptId ON Employees(DeptId);
CREATE UNIQUE INDEX IX_Emp_Email ON Employees(Email);

Clustered vs Non-Clustered

Clustered: Physically sorts data (only 1 per table, usually PK).
Non-Clustered: Separate structure pointing to data (can have multiple).

💡 Use indexes on columns used in WHERE, JOIN, ORDER BY. Avoid on columns with frequent INSERT/UPDATE.

8. Triggers

★ Good to Know 
CREATE TRIGGER trg_AuditSalary
ON Employees
AFTER UPDATE
AS
BEGIN
    INSERT INTO SalaryAudit (EmpId, OldSalary, NewSalary, ChangedDate)
    SELECT d.EmpId, d.Salary, i.Salary, GETDATE()
    FROM deleted d
    JOIN inserted i ON d.EmpId = i.EmpId
    WHERE d.Salary != i.Salary;
END;

9. Transactions

★★ Frequent 
BEGIN TRANSACTION;
BEGIN TRY
    UPDATE Employees SET Salary = Salary + 5000 WHERE EmpId = 1;
    UPDATE Employees SET Salary = Salary - 5000 WHERE EmpId = 2;
    COMMIT;
END TRY
BEGIN CATCH
    ROLLBACK;
    PRINT ERROR_MESSAGE();
END CATCH;
💡 ACID properties: Atomicity, Consistency, Isolation, Durability — frequently asked in theory rounds.
